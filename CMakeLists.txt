cmake_minimum_required(VERSION 3.20)
option(PLUGIN_STANDARD_QFUZZYRESF "Build qFuzzyRESF Scan-to-BIM plugin" OFF)
if(PLUGIN_STANDARD_QFUZZYRESF)
    if(NOT COMMAND AddPlugin)
        message(FATAL_ERROR "Build inside CloudCompare/plugins/private/qFuzzyRESF (Qt 6). See docs/BUILD.md")
    endif()
    project(qFuzzyRESF LANGUAGES CXX)
    AddPlugin(NAME qFuzzyRESF)
    target_sources(qFuzzyRESF PRIVATE
        include/qFuzzyRESF.h include/FuzzyDialog.h
        src/qFuzzyRESF.cpp src/FuzzyDialog.cpp)
    target_include_directories(qFuzzyRESF PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_compile_features(qFuzzyRESF PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(qFuzzyRESF PRIVATE /utf-8)
    else()
        target_compile_options(qFuzzyRESF PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
